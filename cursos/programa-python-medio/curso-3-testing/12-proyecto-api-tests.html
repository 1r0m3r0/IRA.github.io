Suite de Tests: API de Tareas

Tests completos para API CRUD de tareas con pytest + httpx:

# tests/test_tareas.py from httpx import AsyncClient, ASGITransport import pytest from app import app @pytest.fixture def client(): return AsyncClient( transport=ASGITransport(app=app), base_url="http://test" ) @pytest.mark.asyncio async def test_crear_tarea(client): response = await client.post( "/tareas", json={"titulo": "Test", "completada": False} ) assert response.status_code == 201 data = response.json() assert data["titulo"] == "Test" assert "id" in data
