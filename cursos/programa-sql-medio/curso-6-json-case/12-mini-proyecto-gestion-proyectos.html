Disena una BD de gestion de proyectos:

  • JSONB: metadatos flexibles de tareas
  • CASE: categorizar por estado/prioridad
  • Relaciones: 1:N (proyecto-tareas), N:M (empleados-tareas)
CREATE TABLE proyectos ( id SERIAL PRIMARY KEY, nombre TEXT NOT NULL, metadata JSONB DEFAULT '{}' ); CREATE TABLE tareas ( id SERIAL PRIMARY KEY, proyecto_id INT REFERENCES proyectos, titulo TEXT, estado TEXT, prioridad INT );
