Filtro de Eventos Económicos
MQL5: EconomicCalendar API
#include <Trade\Trade.mqh>
// Obtener eventos económicos
EconomicCalendarSelect(
  FROM_DATE, TO_DATE, SYMBOLS_ALL
);
int count = EconomicCalendarTotal();
for(int i = 0; i < count; i++) {
  EconomicCalendarGet(i, event);
  if(event.impact == 1) {
   Print("Evento de alto impacto: ",
     event.event, event.currency);
  }
}
