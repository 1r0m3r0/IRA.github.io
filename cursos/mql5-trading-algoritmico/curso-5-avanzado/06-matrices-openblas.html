// OpenBLAS se activa automáticamente en MQL5 build 2680+
// Las funciones matrix/vector usan OpenBLAS cuando están disponibles
matrix<double> A = {{1,2},{3,4}};
matrix<double> B = {{5,6},{7,8}};
matrix<double> C = A.MatMul(B); // Usa cblas_dgemm internamente
Print("C[0][0]=", C[0][0]);
