SQL Console

Escribe cualquier consulta SQL sobre la base de datos simulada:

sqlite>
-- Resultados aparecerán aquí --
// DatabasePrepare + DatabaseExecute + DatabaseRead
int db = DatabaseOpen("trading_log.db", DATABASE_OPEN_READWRITE);
int stmt = DatabasePrepare(db, "SELECT * FROM trades WHERE profit > 0");
while(DatabaseRead(stmt)) {
   string symbol = DatabaseGetString(stmt, 1);
   double profit = DatabaseGetDouble(stmt, 5);
   Print(symbol, " - Profit: ", profit);
}
DatabaseFinalize(stmt);
DatabaseClose(db);
Trade Logger Simulator

Genera trades simulados y se insertan automáticamente en la base de datos:

// Insertar un trade en MQL5
int db = DatabaseOpen("trading_log.db", DATABASE_OPEN_READWRITE);
int stmt = DatabasePrepare(db, "INSERT INTO trades (symbol,volume,price,profit) VALUES (?,?,?,?)");
DatabaseBind(stmt, 0, "BTCUSD");
DatabaseBind(stmt, 1, 0.1);
DatabaseBind(stmt, 2, 45000.0);
DatabaseBind(stmt, 3, 250.0);
DatabaseExecute(stmt);
DatabaseFinalize(stmt);
DatabaseClose(db);
