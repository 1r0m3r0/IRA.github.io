    _mint(msg.sender, SWORD, 50, "");
    _mint(msg.sender, SHIELD, 30, "");
    _mint(msg.sender, TROPHY, 1, ""); // Ãšnico!
  }

  // Transferencia batch â€” 1 tx para múltiples tokens
  // Gas: ~35% más barato que ERC-20 + ERC-721 separados
  function equipSet(address player) external {
    uint256[] memory ids  = new uint256[](3);
    uint256[] memory amounts = new uint256[](3);
    ids[0]=GOLD; amounts[0]=100;
    ids[1]=SWORD; amounts[1]=1;

      msg.sender, player, ids, amounts, ""

  }

pragma solidity ^0.8.20;
import "@openzeppelin/contracts/token/ERC20/extensions/ERC4626.sol";

contract YieldVault is ERC4626 {
  uint256 public apy; // APY en basis points (1% = 100)

  constructor(IERC20 _asset, uint256 _apy)
    ERC4626(_asset)
    ERC20("Yield USDC", "yUSDC") {
    apy = _apy;
  }

  // Shares-to-assets aumenta con el tiempo (yield)
  function totalAssets() public view override
    returns (uint256) {
    // Los assets crecen con el tiempo + yield
    return IERC20(asset()).balanceOf(address(this))
      + accruedYield;
  }

  // Clave: previewRedeem calcula cuántos assets
  // recibirás por tus shares al retirar
  // previewRedeem(100 shares) = 103 USDC (con yield)
}
